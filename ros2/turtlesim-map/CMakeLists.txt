cmake_minimum_required(VERSION 3.16)
project(rbussi_turtlesim_map)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rcl_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
find_package(turtlesim REQUIRED)
include_directories(include ${Qt5Widgets_INCLUDE_DIRS})
qt5_wrap_cpp(MOC_SOURCES include/turtlesim/turtle_frame.hpp)
add_executable(turtlesim_node src/turtlesim.cpp src/turtle.cpp src/turtle_frame.cpp ${MOC_SOURCES})
ament_target_dependencies(turtlesim_node
  ament_index_cpp geometry_msgs rcl_interfaces rclcpp rclcpp_action std_srvs turtlesim)
target_link_libraries(turtlesim_node Qt5::Widgets)
